cmake_minimum_required(VERSION 3.20)
project(meridian_policy_tests LANGUAGES CXX)
add_executable(meridian_policy_tests policy_test.cpp)
add_executable(meridian_view_tests view_test.cpp)
target_compile_features(meridian_policy_tests PRIVATE cxx_std_20)
target_include_directories(meridian_policy_tests PRIVATE ../../src/platform_se/skyrim_platform)
target_compile_features(meridian_view_tests PRIVATE cxx_std_20)
target_include_directories(meridian_view_tests PRIVATE ../../src/platform_se/skyrim_platform)
if(MSVC)
  target_compile_options(meridian_policy_tests PRIVATE /W4 /WX /utf-8)
  target_compile_options(meridian_view_tests PRIVATE /W4 /WX /utf-8)
endif()
enable_testing()
add_test(NAME meridian_policy COMMAND meridian_policy_tests)
add_test(NAME meridian_public_api_boundary COMMAND meridian_view_tests)
